## 1. `lib/domain/models/item_situation.dart`

```dart
/// Situation of a registered object. Values in English (RNF02);
/// labels in Portuguese belong to the UI.
enum ItemSituation { lost, found }

extension ItemSituationX on ItemSituation {
  /// Portuguese label displayed in the UI (RNF01).
  String get label => switch (this) {
        ItemSituation.lost => 'Perdido',
        ItemSituation.found => 'Encontrado',
      };

  /// Value persisted in the database (`check (situation in ('lost','found'))`).
  String get storageValue => switch (this) {
        ItemSituation.lost => 'lost',
        ItemSituation.found => 'found',
      };

  static ItemSituation fromStorage(String value) => switch (value) {
        'found' => ItemSituation.found,
        _ => ItemSituation.lost,
      };
}
```

## 2. `lib/domain/models/found_item.dart`

```dart
import 'item_situation.dart';

/// Record of a lost or found object.
class FoundItem {
  const FoundItem({
    this.id,
    required this.name,
    required this.description,
    required this.location,
    required this.situation,
    required this.createdAt,
  });

  /// `null` while not persisted; filled by the database after insert.
  final String? id;
  final String name;
  final String description;
  final String location;
  final ItemSituation situation;
  final DateTime createdAt;

  factory FoundItem.fromJson(Map<String, dynamic> json) => FoundItem(
        id: json['id'] as String?,
        name: json['name'] as String,
        description: json['description'] as String,
        location: json['location'] as String,
        situation: ItemSituationX.fromStorage(json['situation'] as String),
        createdAt: DateTime.parse(json['created_at'] as String),
      );

  /// Insert payload. Does **not** send `created_at`: the database generates
  /// date and time automatically (`default now()`), as required by RF02.
  Map<String, dynamic> toInsertJson() => {
        'name': name,
        'description': description,
        'location': location,
        'situation': situation.storageValue,
      };
}
```

## 3. `lib/domain/repositories/item_repository.dart`

Abstract contract (enables testing and backend swaps):

```dart
import '../models/found_item.dart';

/// Access contract for records. All operations are asynchronous (RF03).
abstract interface class ItemRepository {
  /// Records sorted newest first (RF01).
  Future<List<FoundItem>> fetchItems();

  /// Inserts a new record; `created_at` is generated by the database.
  Future<void> addItem(FoundItem item);
}
```

## 4. `lib/data/repositories/supabase_item_repository.dart`

```dart
import 'package:supabase_flutter/supabase_flutter.dart';

import '../../domain/models/found_item.dart';
import '../../domain/repositories/item_repository.dart';

/// Repository implementation backed by Supabase (RF05).
class SupabaseItemRepository implements ItemRepository {
  SupabaseItemRepository({SupabaseClient? client})
      : _client = client ?? Supabase.instance.client;

  final SupabaseClient _client;
  static const String _table = 'items';

  @override
  Future<List<FoundItem>> fetchItems() async {
    final rows = await _client
        .from(_table)
        .select()
        .order('created_at', ascending: false);
    return rows
        .map(FoundItem.fromJson)
        .toList(growable: false);
  }

  @override
  Future<void> addItem(FoundItem item) async {
    await _client.from(_table).insert(item.toInsertJson());
  }
}
```

> Database columns are `snake_case`; model fields are `camelCase`. The mapping
> happens **only** in `fromJson`/`toInsertJson` — never scatter column strings
> through the UI.
